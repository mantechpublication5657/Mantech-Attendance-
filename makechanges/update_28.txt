#TOP PROGRESSING BAR (base.html)

<!-- In <head> -->
<link rel="stylesheet" href="https://cdnjs.cloudflare.com/ajax/libs/nprogress/0.2.0/nprogress.min.css"/>

<!-- Before </body> -->
<script src="https://cdnjs.cloudflare.com/ajax/libs/nprogress/0.2.0/nprogress.min.js"></script>
<script>
  NProgress.configure({ showSpinner: false, speed: 400, trickleSpeed: 200 });

  // Start bar on every link click and form submit
  document.addEventListener('click', function(e) {
    const link = e.target.closest('a[href]');
    if (link && !link.target && !link.href.startsWith('#') && !link.href.startsWith('javascript')) {
      NProgress.start();
    }
  });

  document.addEventListener('submit', function() {
    NProgress.start();
  });

  // Finish when page fully loads
  window.addEventListener('load', function() {
    NProgress.done();
  });
</script>
